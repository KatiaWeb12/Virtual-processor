SHOWWINDOW

; Center: (400, 300)
; RX = 300, RY = 150
; Image: 800 x 600

; RAX = X
; RBX = Y
; RCX = DX / RAM address
; RDX = DY

; Y = 150
PUSH 150
POP RBX

:y_loop

; X = 100
PUSH 100
POP RAX

:x_loop

; DX = X - 400
PUSH 400
PUSH RAX
SUB
POP RCX

; DY = Y - 300
PUSH 300
PUSH RBX
SUB
POP RDX

; DX^2
PUSH RCX
PUSH RCX
MUL

; 4 * DY^2
PUSH RDX
PUSH RDX
MUL
PUSH 10
MUL

; DX^2 + 4 * DY^2
ADD

; Check DX^2 + 4 * DY^2 <= 90000
PUSH 80000
JAE :draw_pixel

JMP :next

:draw_pixel

; RAM address = Y * 800 + X
PUSH RBX
PUSH 800
MUL
PUSH RAX
ADD
POP RCX

; RAM[address] = 1
PUSH 1
POP [RCX]

:next

; X += 1
PUSH RAX
PUSH 1
ADD
POP RAX

; Continue while X <= 700
PUSH 700
PUSH RAX
JBE :x_loop

; Y += 1
PUSH RBX
PUSH 1
ADD
POP RBX

; Continue while Y <= 450
PUSH 450
PUSH RBX
JBE :y_loop

; Draw current row
DRAW

HLT
