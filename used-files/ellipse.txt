SHOWWINDOW
; Center: (400, 300)
; Radius: 200
; Image 800 * 600

; RAX = X
; RBX = Y
; RCX = DX / RAM address
; RDX = DY

; Y = 100
PUSH 100
POP RBX

:y_loop

; X = 200
PUSH 200
POP RAX

:x_loop

; DX = X - 400
PUSH 400
PUSH RAX
SUB
POP RCX

; DY = Y - 300
PUSH 300
PUSH RBX
SUB
POP RDX

; DX^2
PUSH RCX
PUSH RCX
MUL

; DY^2
PUSH RDX
PUSH RDX
MUL

; DX^2 + DY^2
ADD

; Check DX^2 + DY^2 <= R^2(40000)
PUSH 40000
JAE :draw

JMP :next

:draw

; RCX = RAM address = Y * 800 + X
PUSH RBX
PUSH 800
MUL
PUSH RAX
ADD
POP RCX

; RAM[address] = 1
PUSH 1
POP [RCX]

:next

; X += 1
PUSH RAX
PUSH 1
ADD
POP RAX

; Continue while X <= 600
PUSH 600
PUSH RAX
JBE :x_loop

; Draw the completed row
DRAW

; Y += 1
PUSH RBX
PUSH 1
ADD
POP RBX

; Continue while Y <= 500
PUSH 500
PUSH RBX
JBE :y_loop

HLT
