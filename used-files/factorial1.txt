IN

CALL :factorial

OUT
HLT

:factorial
    POP RAX
    PUSH RBX        ; кладём RBX на стек

    PUSH 1
    PUSH RAX
    JBE :base

    PUSH RAX
    POP RBX         ; сохраняем n в RBX

    PUSH 1
    PUSH RAX        ; вычисляем n - 1
    SUB

    CALL :factorial
    PUSH RBX
    MUL

    POP RAX
    POP RBX     ; восстанавливаем предыдущее значение RBX (из строки 10)
    PUSH RAX    ; возвращаем результат в стек
    RET

:base
    POP RBX
    PUSH 1
    RET
